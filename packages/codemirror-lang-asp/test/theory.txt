# Theory atom with guard

&diff { x - y } <= -3 :- p.

==>

SourceFile(Rule(TheoryAtom("&",Identifier,"{",TheoryElement(TheoryFunction(Identifier),TheoryOperator,TheoryFunction(Identifier)),"}",TheoryOperator,TheoryOperator,Number),":-",Body(SymbolicAtom(Identifier)),"."))

# Theory atom elements

&sum { x; 1 : p; : q }.

==>

SourceFile(Rule(TheoryAtom("&",Identifier,"{",TheoryElement(TheoryFunction(Identifier)),";",TheoryElement(Number,":",Condition(SymbolicAtom(Identifier))),";",TheoryElement(":",Condition(SymbolicAtom(Identifier))),"}"),"."))

# Theory atom arguments

&dom(1) { 1..3 }.

==>

SourceFile(Rule(TheoryAtom("&",Identifier,"(",Number,")","{",TheoryElement(Number,TheoryOperator,Number),"}"),"."))

# Theory terms

&p { f(a, b), (), (1,), (1, 2), [1, 2], {1}, "s", #inf, X }.

# Theory terms beyond clingo 5.8

&p { (,), _ }.

==>

# Theory operators and not

==>

SourceFile(Rule(TheoryAtom("&",Identifier,"{",TheoryElement(not,TheoryFunction(Identifier),TheoryOperator,TheoryOperator,TheoryFunction(Identifier),TheoryOperator,TheoryFunction(Identifier)),"}"),"."))

# Theory atom without elements

==>
